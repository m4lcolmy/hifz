# Python 3.12. Versions are the ones this has been tested against; they are
# pinned loosely (>=) so patch updates are fine.

# ── Interface ──────────────────────────────────────────────────────────
PyQt6>=6.11.0                 # window, Mushaf rendering, microphone capture

# ── Speech recognition ─────────────────────────────────────────────────

# ── Tests ──────────────────────────────────────────────────────────────
pytest>=9.0.0
